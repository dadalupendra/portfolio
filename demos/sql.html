📝 SQL Queries

-- SELECT with JOIN: Get students and their courses
SELECT s.name, c.course_name, e.grade
FROM students s
JOIN enrollments e ON s.student_id = e.student_id
JOIN courses c ON c.course_id = e.course_id
WHERE s.country = 'Nepal'
ORDER BY c.course_name;
-- GROUP BY: Count students per country
SELECT country, COUNT(*) AS student_count
FROM students
GROUP BY country
ORDER BY student_count DESC;
-- PL/SQL: Stored procedure to enroll a student
CREATE OR REPLACE PROCEDURE enroll_student(
    p_student_id IN NUMBER,
    p_course_id  IN NUMBER
) AS
BEGIN
    INSERT INTO enrollments (student_id, course_id, grade)
    VALUES (p_student_id, p_course_id, 'IP');
    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Student enrolled successfully!');
END;
