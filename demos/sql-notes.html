📌 2. JOIN — Multiple Tables

Combine students with their course enrollments.

SELECT s.name, c.course_name, e.grade
FROM students s
INNER JOIN enrollments e ON s.id = e.student_id
INNER JOIN courses c ON c.id = e.course_id
ORDER BY s.name;

📌 3. GROUP BY & Aggregates

Count students per country and find averages.

SELECT country, COUNT(*) AS total,
       ROUND(AVG(score), 1) AS avg_score
FROM students
GROUP BY country
HAVING COUNT(*) >= 2
ORDER BY avg_score DESC;

📌 4. Subquery

Find students with above-average scores.

SELECT name, score
FROM students
WHERE score > (SELECT AVG(score) FROM students)
ORDER BY score DESC;

📌 5. CREATE VIEW

Create a reusable view for common queries.

CREATE VIEW student_summary AS
SELECT s.name, s.country,
       COUNT(e.course_id) AS courses_taken,
       ROUND(AVG(e.score), 1) AS avg_grade
FROM students s
LEFT JOIN enrollments e ON s.id = e.student_id
GROUP BY s.name, s.country;

📌 6. PL/SQL Procedure

A stored procedure with error handling.

CREATE OR REPLACE PROCEDURE add_student(
    p_name IN VARCHAR2,
    p_email IN VARCHAR2,
    p_country IN VARCHAR2
) AS
BEGIN
    INSERT INTO students (name, email, country, enrollment_date)
    VALUES (p_name, p_email, p_country, SYSDATE);
    COMMIT;
    DBMS_OUTPUT.PUT_LINE('Student added: ' || p_name);
EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN
        DBMS_OUTPUT.PUT_LINE('Error: Email already exists');
END;
