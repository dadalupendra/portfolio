🔄 Loops & Logic

# FizzBuzz — classic coding challenge
for i in range(1, 16):
    if i % 15 == 0:
        print("FizzBuzz")
    elif i % 3 == 0:
        print("Fizz")
    elif i % 5 == 0:
        print("Buzz")
    else:
        print(i)

🔧 Functions

# Temperature converter function
def celsius_to_fahrenheit(celsius):
    """Convert Celsius to Fahrenheit"""
    return (celsius * 9/5) + 32

# Nepal city temperatures
cities = {"Kathmandu": 22, "Pokhara": 25, "Lumbini": 30}

for city, temp in cities.items():
    f_temp = celsius_to_fahrenheit(temp)
    print(f"{city}: {temp}°C = {f_temp}°F")
Kathmandu: 22°C = 71.6°F
Pokhara: 25°C = 77.0°F
Lumbini: 30°C = 86.0°F
